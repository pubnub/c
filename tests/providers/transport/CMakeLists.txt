# curl transport -- always built against the selected transport
# target (which maps to the libcurl implementation when
# PUBNUB_PROVIDER_TRANSPORT=curl) rather than pulling the source
# directly.  That way the test exercises the same binary consumers
# will see.  The test also needs the allocator (response buffers
# are acquired via PUBNUB_BUF_RX at send time), so link the
# stdlib allocator target when it is the active backend.
if(PUBNUB_PROVIDER_TRANSPORT STREQUAL "curl")
    pubnub_add_unit_test(
        test_provider_transport_curl_units
        SOURCES curl_transport_units.c
        LIBS pubnub
    )
endif()

# Socket transport compression utility tests (gzip/deflate inflate)
if(PUBNUB_ENABLE_COMPRESSION)
    if(PUBNUB_PROFILE STREQUAL "embedded")
        # tinfl (embedded) backend — uses pre-baked fixtures, no system zlib needed
        pubnub_resolve_miniz()
        pubnub_add_unit_test(
            test_provider_transport_socket_compression_tinfl
            SOURCES socket/test_compression_tinfl.c ${PN_MINIZ_SOURCE_DIR}/miniz_tinfl.c
        )
        target_include_directories(
            test_provider_transport_socket_compression_tinfl
            PRIVATE
                ${PN_PROVIDERS_SRC_DIR}/transport/socket/inflate
                ${CMAKE_CURRENT_SOURCE_DIR}/socket
                ${PN_MINIZ_SOURCE_DIR}
        )
        target_compile_definitions(
            test_provider_transport_socket_compression_tinfl
            PRIVATE PUBNUB_COMPRESSION_BACKEND_TINFL
        )
        # miniz_export.h is generated by miniz's CMake build; when compiling
        # miniz_tinfl.c directly, point the compiler at a stub header in the
        # test directory that defines MINIZ_EXPORT as empty.
        target_include_directories(
            test_provider_transport_socket_compression_tinfl
            BEFORE
            PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/socket
        )
    else()
        find_package(ZLIB QUIET)
        if(NOT ZLIB_FOUND)
            include(FetchContent)
            FetchContent_Declare(
                zlib_upstream
                GIT_REPOSITORY https://github.com/madler/zlib.git
                # Pinned to immutable commit SHA (tags are mutable / repointable).
                GIT_TAG
                    51b7f2abdade71cd9bb0e7a373ef2610ec6f9daf # v1.3.1
                GIT_SHALLOW TRUE
            )
            FetchContent_MakeAvailable(zlib_upstream)
            pubnub_add_unit_test(
                test_provider_transport_socket_compression
                SOURCES
                    socket/test_compression.c
                    ${PN_PROVIDERS_SRC_DIR}/transport/socket/inflate/pn_inflate.c
                LIBS zlibstatic
            )
            target_include_directories(
                test_provider_transport_socket_compression
                PRIVATE "${zlib_upstream_SOURCE_DIR}" "${zlib_upstream_BINARY_DIR}"
            )
        else()
            pubnub_add_unit_test(
                test_provider_transport_socket_compression
                SOURCES
                    socket/test_compression.c
                    ${PN_PROVIDERS_SRC_DIR}/transport/socket/inflate/pn_inflate.c
                LIBS ZLIB::ZLIB
            )
        endif()
        target_compile_definitions(
            test_provider_transport_socket_compression
            PRIVATE PUBNUB_COMPRESSION_BACKEND_ZLIB
        )
    endif()
endif()

    # DNS resolver integration test requires pthreads (mock server) and POSIX
    # socket ops included directly — not buildable on Windows.
    # Only built when custom DNS is enabled (exercises the UDP resolver path).
    if(PUBNUB_ENABLE_CUSTOM_DNS AND NOT WIN32)
        pubnub_add_unit_test(
            test_provider_transport_socket_dns_resolver
            SOURCES socket/test_dns_resolver.c socket/mock/mock_dns_server.c
            LIBS pubnub
        )
        target_include_directories(
            test_provider_transport_socket_dns_resolver
            PRIVATE ${PN_PROVIDERS_SRC_DIR}/transport/socket/dns ${CMAKE_CURRENT_SOURCE_DIR}/socket
        )
        # platform_posix.c is included directly; give it the same feature-test macros
        # that src/providers/platform/posix/CMakeLists.txt sets on the library target.
        # _DARWIN_C_SOURCE is required on macOS to keep BSD symbols (INADDR_LOOPBACK
        # etc.) visible when _POSIX_C_SOURCE is explicitly set.
        target_compile_definitions(
            test_provider_transport_socket_dns_resolver
            PRIVATE
                _POSIX_C_SOURCE=200809L

        )

        # Regression guard for index-based DNS server-rotation termination

    # test_decompression.c white-boxes transport_socket.c via direct #include to
    # reach the file-static decompression helpers, and mocks the inflate backend
    # so the output buffer capacity and the inflated length are observably
    # different. Links no SDK libraries: every external dependency is stubbed in

    pubnub_add_unit_test(
        test_provider_transport_socket_decompression

        test_provider_transport_socket_decompression
        PRIVATE ${PN_PROVIDERS_SRC_DIR}/transport/socket
    )
    # On Windows, transport_socket.c references WSAStartup/WSACleanup
    # (ws2_32) and pn_windows_socket_ops (defined in windows_socket_ops.c).
    # The test stubs everything else; supply only the missing platform symbols.
    if(WIN32)
        target_sources(
            test_provider_transport_socket_decompression
            PRIVATE ${PN_PROVIDERS_SRC_DIR}/transport/socket/platform/windows_socket_ops.c
        )
        target_link_libraries(test_provider_transport_socket_decompression PRIVATE ws2_32)
